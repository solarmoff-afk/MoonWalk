# Архитектура MoonWalk
Дерево основного крейта **moonwalk**

```
├── batching                    Система пакетной отрисовки
│   ├── common.rs               Основные методы батчинга, такие как сортировка по z индексу
│   ├── group.rs                Группа батчинга
│   ├── mod.rs                  Файл модуля
│   └── shapes                  Конкретные виды батчинга (на будущее)
│       ├── mod.rs              Файл модуля
│       └── uber.rs             Единый батчинг для текста и прямоугольников
├── debug.rs                    Макросы для дебага/профилирования
├── error.rs                    Реализация ошибок через крейт thiserror
├── filters                     Система фильтров для обработки текстур
│   ├── color_matrix.rs         Хелперы для удобной цветокорекции
│   ├── factory.rs              Фабрика конвейеров рендеринга для обработки текстур
│   ├── mod.rs                  Файл модуля
│   ├── shaders                 Шейдеры фильтров
│   │   ├── advanced.wgsl       Хромокей и маска
│   │   ├── blur.wgsl           Блюр по Гауссу
│   │   └── color_matrix.wgsl   Матрица цвета (цветокорекция)
│   └── uniforms.rs             Юниформы для шейдеров
├── gpu                         В прошлом абстрации над wgpu, сейчас наследие и матричный стэк
│   ├── matrix.rs               Матричный стэк в стиле старого OpenGL
│   └── mod.rs                  Файл модуля
├── lib.rs                      Файл крейта
├── objects                     Модуль для работы с объектами
│   ├── mod.rs                  Файл модуля
│   └── store.rs                Храниилище всех объектов на SOA архитектуре с кэшированием
├── painting                    Модуль для рисования растровыми кистями
│   └── mod.rs                  Файл модуля
├── path                        Модуль для рисования векторными кистями
│   ├── mod.rs                  Файл модуля
│   └── svg.rs                  Парсинг SVG
├── public                      Публичный api
│   ├── brush.rs                Api растровых кистей
│   ├── custom.rs               Api CustomPaint
│   ├── export.rs               Api для экспорта (рендер, сохранение текстуры на диск)
│   ├── filters.rs              Api для фильтров
│   ├── getter.rs               Api для получения значений
│   ├── mod.rs                  Файл модуля
│   ├── objects.rs              Api для работы с объктами
│   ├── resources.rs            Api для загрузки и работы с текстурами и шрифтами
│   └── video.rs                Api для работы с видео рекордером
├── rendering                   Модуль для рендеринга
│   ├── container.rs            Оффскрин рендеринг
│   ├── custom.rs               CustomPaint
│   ├── mod.rs                  Файл модуля
│   ├── pipeline.rs             Основной конвейер рендера с возможностью создавать дополнительные
│   ├── renderer.rs             Структура рендерера
│   ├── snapshot.rs             Обрезка снапшотов
│   ├── state.rs                Состояние рендера, работа с текстурами и батчингом
│   ├── vertex.rs               Структура вершины
│   └── video.rs                Обёртка над ffmpeg через video-rs
├── resource_manager.rs         Загрузка текстур и шрифтов
├── shaders                     Шейдеры
│   ├── brush.wgsl              Шейдер для растровых кистей
│   ├── empty.wgsl              Пустой шейдер (наследие)
│   ├── path.wgsl               Шейдер для векторных путей
│   └── shape.wgsl              Шейдер для прямоугольников и текстов
└── textware                    Модуль для рендеринга текста
    ├── cache.rs                Кэширование глифов
    ├── error.rs                Ошибки текстовой системы
    ├── font.rs                 Работа со шрифтов
    └── mod.rs                  Файл модуля
```
